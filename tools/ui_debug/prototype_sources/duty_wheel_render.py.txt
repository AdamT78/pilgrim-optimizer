#!/usr/bin/env python3
"""Render pilgrim_board.html to a 2x PNG via headless Chromium.

cairosvg is not available in this environment, so the SVG is rasterised
through the browser instead; the <svg> element is screenshotted directly
so the PNG is the board alone at its true aspect.
"""
import asyncio
import sys
from pathlib import Path
from playwright.async_api import async_playwright

HERE = Path(__file__).parent
NAME = sys.argv[1] if len(sys.argv) > 1 else "pilgrim_board"
CHROME = "/opt/pw-browsers/chromium-1194/chrome-linux/chrome"


async def main():
    async with async_playwright() as p:
        browser = await p.chromium.launch(executable_path=CHROME)
        page = await browser.new_page(viewport={"width": 1200, "height": 1500},
                                      device_scale_factor=2)
        await page.goto(f"file://{HERE / (NAME + '.html')}")
        await page.wait_for_timeout(350)
        el = await page.query_selector("svg")
        await el.screenshot(path=str(HERE / f"{NAME}.png"))
        await browser.close()
    print(f"wrote {NAME}.png")


asyncio.run(main())
