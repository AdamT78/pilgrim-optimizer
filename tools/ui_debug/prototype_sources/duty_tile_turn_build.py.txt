def duty_text() -> dict:
    """gen_board.py's DUTY_TEXT, read as a literal rather than by running the module.

    It is the one table that names the ACTIONS -- "Devotion", "Silversmith" -- for all thirteen
    hit areas, including the three tiles that carry only one. Parsing the literal keeps this page
    out of gen_board's import side effects, which include writing a page of its own.
    """
    src = (REPO / "ui" / "render" / "gen_board.py").read_text(encoding="utf-8")
    tree = ast.parse(src)
    for node in tree.body:
        if isinstance(node, ast.Assign) and any(
                getattr(t, "id", "") == "DUTY_TEXT" for t in node.targets):
            return ast.literal_eval(node.value)
    raise SystemExit("gen_board.py no longer assigns DUTY_TEXT at module level")


def white_tints() -> dict[str, str]:
    """`acolyte_tints`, with the dark silhouette rebuilt in white."""
    import colorsys

    import numpy as np
    from PIL import Image, ImageFilter

    src = Image.open(dg.ACOLYTE).convert("RGBA")
    out = {}
    for seat, hexv in dg.SEAT_SWATCH.items():
        r, g, b = (int(hexv[i:i + 2], 16) / 255 for i in (1, 3, 5))
        h, s, v = colorsys.rgb_to_hsv(r, g, b)
        dark = colorsys.hsv_to_rgb(h, min(1, s * 1.3), v * 0.30)
        light = colorsys.hsv_to_rgb(h, min(1, s * 0.95), min(1, v * 1.10))
        a = np.asarray(src).astype(float)
        lum = (0.2126 * a[..., 0] + 0.7152 * a[..., 1] + 0.0722 * a[..., 2]) / 255.0
        lum = np.clip((lum - 0.06) / 0.84, 0, 1) ** 0.80
        o = np.zeros_like(a)
        for c in range(3):
            o[..., c] = (dark[c] + (light[c] - dark[c]) * lum) * 255
        o[..., 3] = a[..., 3]
        fig = Image.fromarray(o.astype("uint8"), "RGBA")
        alpha = np.asarray(fig)[..., 3]
        halo = Image.fromarray(alpha).filter(ImageFilter.MaxFilter(13))
        sil = Image.new("RGBA", fig.size, HALO + (255,))
        sil.putalpha(halo)
        base = Image.new("RGBA", fig.size, (0, 0, 0, 0))
        base.alpha_composite(sil)
        base.alpha_composite(fig)
        buf = io.BytesIO()
        base.save(buf, "PNG")
        out[seat] = "data:image/png;base64," + base64.b64encode(buf.getvalue()).decode("ascii")
    return out


def pointlist(d: str) -> list[list[float]]:
    return [[float(m.group(1)), float(m.group(2))]
            for m in re.finditer(r"(-?\d+\.?\d*)[ ,]+(-?\d+\.?\d*)", d)]


def as_path(pts) -> str:
    return "M " + " L ".join("%.1f %.1f" % (x, y) for x, y in pts) + " Z"


def rect_grid(box: float) -> list[str]:
    """Nine identical squares, evenly spaced. Same path grammar as the traced shapes."""
    side = (box - 2 * RECT_MARGIN - 2 * RECT_CHANNEL) / 3.0
    out = []
    for i in range(9):
        c, r = i % 3, i // 3
        x = RECT_MARGIN + c * (side + RECT_CHANNEL)
        y = RECT_MARGIN + r * (side + RECT_CHANNEL)
        out.append(as_path([[x, y], [x + side, y], [x + side, y + side], [x, y + side]]))
    return out


def tile_shapes() -> list[str]:
    """The nine outlines this page draws, whichever kind they are.

    ONE function, called by the page and by the checks below it. They read from the same place on
    purpose: a check that rebuilds its own copy of the thing it is checking is a check that can
    pass while the page is wrong, and this file has already had that argument once.
    """
    if SHAPES_ARE == "rect":
        return rect_grid(dg.load()["box"])
    return [grow(flat_base(d), GROW) for d in dg.laid_shapes(pop_set=POP_SET)]


def flat_base(d: str) -> str:
    """Push the tile's scalloped bottom down onto one straight line at its lowest point."""
    pts = pointlist(d)
    ys = [p[1] for p in pts]
    xs = [p[0] for p in pts]
    y0, y1, x0, x1 = min(ys), max(ys), min(xs), max(xs)
    lo, hi = x0 + (x1 - x0) * 0.10, x1 - (x1 - x0) * 0.10
    bottom = [y for x, y in pts if lo <= x <= hi and y > y0 + (y1 - y0) * 0.70]
    cut = y1 - ((y1 - min(bottom)) + FLAT_PAD) if bottom else y1
    for pt in pts:
        if pt[1] > cut:
            pt[1] = y1
    return as_path(pts)


def grow(d: str, k: float) -> str:
    """Scale one outline about its own centre, so it takes room from the channel and moves not."""
    pts = pointlist(d)
    xs = [p[0] for p in pts]
    ys = [p[1] for p in pts]
    cx, cy = (min(xs) + max(xs)) / 2, (min(ys) + max(ys)) / 2
    for pt in pts:
        pt[0] = cx + (pt[0] - cx) * k
        pt[1] = cy + (pt[1] - cy) * k
    return as_path(pts)

def data_uri(path: pathlib.Path) -> str:
    mime = {".webp": "image/webp", ".png": "image/png"}[path.suffix.lower()]
    return "data:%s;base64,%s" % (mime, base64.b64encode(path.read_bytes()).decode("ascii"))


def shade(hexv: str, v: float, sat: float) -> str:
    """One seat colour moved in value and saturation, keeping its hue."""
    import colorsys
    r, g, b = (int(hexv[i:i + 2], 16) / 255 for i in (1, 3, 5))
    h, s_, v_ = colorsys.rgb_to_hsv(r, g, b)
    out = colorsys.hsv_to_rgb(h, min(1.0, s_ * sat), min(1.0, v_ * v))
    return "#" + "".join("%02x" % round(c * 255) for c in out)


def lit(hexv: str) -> str:
    """One colour put through the board's own LIT filter, in sRGB, by hand.

    `gen_duty_grid.LIT` is a saturate of 1.20 followed by a linear transfer per channel, and the
    filters that use it declare color-interpolation-filters="sRGB" -- so this is the same
    arithmetic on the same numbers, not a lighter purple chosen by eye. If the filter is ever
    retuned, this follows it by being re-derived from those constants rather than by someone
    remembering a hex.
    """
    sat = 1.20
    r, g, b = (int(hexv[i:i + 2], 16) / 255 for i in (1, 3, 5))
    rr = (0.213 + 0.787 * sat) * r + (0.715 - 0.715 * sat) * g + (0.072 - 0.072 * sat) * b
    gg = (0.213 - 0.213 * sat) * r + (0.715 + 0.285 * sat) * g + (0.072 - 0.072 * sat) * b
    bb = (0.213 - 0.213 * sat) * r + (0.715 - 0.715 * sat) * g + (0.072 + 0.928 * sat) * b
    out = (rr * 1.34 + 0.10, gg * 1.14 + 0.055, bb * 0.78 + 0.0)
    return "#" + "".join("%02x" % round(min(1.0, max(0.0, c)) * 255) for c in out)


def bbox(d: str):
    xs, ys = [], []
    for m in re.finditer(r"(-?\d+\.?\d*)[ ,]+(-?\d+\.?\d*)", d):
        xs.append(float(m.group(1)))
        ys.append(float(m.group(2)))
    return min(xs), min(ys), max(xs), max(ys)


def safe_top(d: str, span: float = 0.66) -> float:
    """The lowest point of the tile's top edge across its middle -- where text stops sticking out.

    The outlines are traced and wavy, so a title set from the bounding box's top sits outside the
    ink on every tile. Measured per tile because the nine differ.
    """
    x0, y0, x1, y1 = bbox(d)
    lo, hi = x0 + (x1 - x0) * (1 - span) / 2, x1 - (x1 - x0) * (1 - span) / 2
    top = [y for x, y in ((float(m.group(1)), float(m.group(2)))
                          for m in re.finditer(r"(-?\d+\.?\d*)[ ,]+(-?\d+\.?\d*)", d))
           if lo <= x <= hi and y < y0 + (y1 - y0) * 0.40]
    return max(top) if top else y0


# Shorter names for five of the thirteen, asked for after seeing them on the tiles. The tile's
# own title carries the verb -- "Construct" over "Building" and "Road" -- so the action only has
# to say which of the two it is.
#
# This renames the SANDBOX's copy. gen_board.DUTY_TEXT still says the long names, and it is what
# the game view's inspect banner reads, so the two disagree until that file is changed too.
RENAME = {
    "Construct a building": "Building",
    "Construct a road": "Road",
    "Donate a building": "Building",
    "Produce wheat": "Wheat",
    "Produce stone": "Stone",
}


def renamed(text: dict) -> dict:
    """DUTY_TEXT with the five short names applied, and a check that all five were found.

    A rename map is the quietest thing in a codebase to get wrong: a key that matches nothing
    changes nothing, the page still builds, and the old name is still on the tile. "Donate a
    Building" was asked for with a capital B and the table says "Donate a building", so this
    would already have missed one silently.
    """
    out, hit = {}, set()
    for k, v in text.items():
        title = v[0]
        if title in RENAME:
            hit.add(title)
            out[k] = (RENAME[title],) + tuple(v[1:])
        else:
            out[k] = v
    missing = sorted(set(RENAME) - hit)
    if missing:
        raise SystemExit(
            "these action names are not in gen_board.DUTY_TEXT, so renaming them did nothing: %s\n"
            "The table now says: %s"
            % (", ".join(missing), ", ".join(sorted(v[0] for v in text.values()))))
    return out


def span_at(d: str, y: float, band: float = 26.0) -> tuple[float, float]:
    """How wide the outline actually is at one height.

    The bounding box is wider than the tile at every height the captions use, because the traced
    top curves in. Fitting "Silversmith" to the box put it over the edge; fitting it to the ink
    at its own baseline does not. Measured in a band around y, since the outline is a polyline
    and may have no vertex exactly there.
    """
    pts = [(float(m.group(1)), float(m.group(2)))
           for m in re.finditer(r"(-?\d+\.?\d*)[ ,]+(-?\d+\.?\d*)", d)]
    near = [x for x, yy in pts if abs(yy - y) <= band]
    if len(near) < 2:
        x0, _y0, x1, _y1 = bbox(d)
        return x0, x1
    return min(near), max(near)


def actions(text: dict, name: str) -> list[str]:
    """The action name or names on one tile, in the drawing's own order."""
    return [text[k][0] for k in ("%s|0" % name, "%s|1" % name) if k in text]

def build(plan=None, over=None) -> str:
    text = renamed(duty_text())
    # SET FROM THE PRISTINE VALUE, never from the current one. `build` mutates the shared
    # population set, and the moment it was called twice -- once per page -- the shrink applied
    # twice and every figure came out at 0.88 squared. The page still built and still looked
    # plausible; only a number printed at the end said otherwise.
    _t = pop.SETS[POP_SET]["tile"]
    _t["frac"] = FRAC_PRISTINE * FIG_SHRINK
    if EQUAL_QUARTERS:
        if _t["frac"] >= 0.25:
            raise SystemExit(
                "a figure is %.3f of the tile, so four of them cannot fit in four quarters"
                % _t["frac"])
        _t["gap"] = (0.25 - _t["frac"]) / _t["frac"]
    laid = tile_shapes()
    joins = dg.joins_for()
    box = dg.load()["box"]

    # The hood is an OUTLINED SHAPE, so its border is a stroke rather than a dilated halo: white
    # is one value, set where hood_defs reads it. `pop_defs` has to be emitted or the row draws
    # NOTHING -- an unresolved <use> is silent, and a board with no acolytes is one somebody
    # believes. The count printed at the end is what checks it did.
    pop.HOOD_INK = "#ffffff"
    plan = PLAN_NINE if plan is None else plan
    over = over or {}
    o_rects, o_badge, o_lit = (over.get("rects", {}), over.get("no_badge", set()),
                               over.get("lit_top", {}))
    o_mark, o_art = over.get("mark", {}), over.get("art", {})
    o_act, o_plabel = over.get("act_art", {}), over.get("panel_label", {})
    o_gold = over.get("gold", {})
    o_litp, o_square = over.get("lit_piece", {}), over.get("square", set())
    parts = ['<svg viewBox="0 0 %d %d" xmlns="http://www.w3.org/2000/svg">' % (box, box),
             "<defs>%s</defs>" % dg.pop_defs(UID, POP_SET, dg.SEAT_ORDER)]
    for cell, d in enumerate(laid):
        i = plan[cell]
        x0, y0, x1, y1 = bbox(d)
        cx = (x0 + x1) / 2
        top = safe_top(d)
        parts.append('<g class="dgt" data-i="%d">' % cell)
        parts.append('<path class="fillp" d="%s"/>' % d)
        parts.append('<path class="edge" d="%s"/>' % d)
        mk = o_mark.get(cell)
        if mk in ("steady", "pulse"):
            parts.append('<path class="mark mark-%s" d="%s"/>' % (mk, d))
        if i is None:
            # A BARE TILE: the same rectangle and the same top panel at the same depth as its
            # neighbours, and nothing in it. The panel keeps its own default height rather than
            # borrowing whatever the Clericals grew to, so the row of panels still lines up.
            parts.append('<rect class="panel" x="%.1f" y="%.1f" width="%.1f" height="%.1f"/>'
                         % (x0 + PANEL_INSET, y0 + BAND,
                            (x1 - x0) - 2 * PANEL_INSET, ACTION_DY + BLOCK_PAD))
            parts.append("</g>")
            continue
        name = dg.DUTY_NAMES[i]
        acts = actions(text, name)
        # The panel, then the block pushed down by exactly its height. A tile with no panel is
        # not a tile with an empty one: nothing is drawn and nothing is displaced, so its title
        # sits where every title sat before the panel existed.
        # THE TILE, TOP TO BOTTOM: title band, panel, action band, art, acolyte line.
        #
        # Both text bands are the SAME depth and both set their type at TITLE_DY into their own
        # band, so the title above the panel and the actions below it are hung off one number
        # rather than two that have to be kept in agreement. The action band is reserved on every
        # tile including the City, which has no actions to put in it -- the panels sit in a grid
        # and a band that changed depth per tile would stagger the art areas across the row.
        in_panel = name in ACOLYTES_IN_PANEL
        # `in`, not `.get(...) or ...`: an EMPTY override means "this tile draws no pieces",
        # and truthiness cannot tell that apart from "no override", which would silently put the
        # acolytes back.
        squares = o_rects[cell] if cell in o_rects else RECTS.get(name)
        has_rects = squares is not None
        panel_h = ACTION_DY + BLOCK_PAD
        if in_panel:
            m = pop.tile(POP_SET)
            bx = dg.acolyte_box(d, pop_set=POP_SET)
            # The panel is sized to WHAT THIS TILE ACTUALLY DRAWS, not to the acolyte counts it
            # would have drawn. A tile showing squares has no pile, and sizing it to a pile that
            # is not there leaves a panel with a hand's width of nothing under the pieces.
            if has_rects:
                tall = (bx["fh"] * m["step"] * max(squares.values())) if squares else 0.0
            else:
                tall = bx["fh"] + (max(COUNTS[i]) - 1) * bx["fh"] * m["step"]
            panel_h = max(panel_h, tall + PANEL_PILE_PAD)

        title_y = y0 + TITLE_DY
        panel_top = y0 + BAND
        panel_bot = panel_top + (panel_h if name not in NO_PANEL else 0.0)
        row_y = panel_bot + ACTION_DROP
        art_top = panel_bot + BAND

        t_l, t_r = span_at(d, title_y)
        t_size = min(30.0, (t_r - t_l - 20) / (len(name) * 0.52))
        if cell in o_square:
            # Squared off against the TITLE'S OWN CAP HEIGHT rather than the band's, so it sits
            # on the same optical line as the word beside it. A 30-unit serif caps at about 0.7
            # of its size; the square matches that and is centred on it.
            cap = t_size * 0.70
            parts.append('<rect class="slot" x="%.1f" y="%.1f" width="%.1f" height="%.1f"/>'
                         % (x0 + PANEL_INSET, title_y - cap, cap, cap))
        parts.append('<text class="tname" x="%.1f" y="%.1f" font-size="%.1f">%s</text>'
                     % ((t_l + t_r) / 2, title_y, t_size, name))
        badge = None if cell in o_badge else BADGE.get(name)
        if badge:
            # Right-anchored on the title's own baseline, so the two sit on one row whatever the
            # title's width does -- the title is centred and shrinks to fit, and anchoring this
            # to the title's box instead would move it every time a name got longer.
            parts.append('<text class="tval" x="%.1f" y="%.1f">%s</text>'
                         % (t_r - BADGE_INSET, title_y, badge))
        if name not in NO_PANEL:
            px, pw = x0 + PANEL_INSET, (x1 - x0) - 2 * PANEL_INSET
            art = o_art.get(cell)
            if art is not None and art.is_file():
                cid = "art%d" % cell
                parts.append('<clipPath id="%s"><rect x="%.1f" y="%.1f" width="%.1f" '
                             'height="%.1f"/></clipPath>' % (cid, px, panel_top, pw, panel_h))
                parts.append('<image href="%s" x="%.1f" y="%.1f" width="%.1f" height="%.1f" '
                             'preserveAspectRatio="xMidYMid slice" clip-path="url(#%s)"/>'
                             % (data_uri(art), px, panel_top, pw, panel_h, cid))
                parts.append('<rect class="panel panel-art" x="%.1f" y="%.1f" width="%.1f" '
                             'height="%.1f"/>' % (px, panel_top, pw, panel_h))
            label = o_plabel.get(cell)
            if label:
                # Centred across the panel, and set at the SAME drop from its box's top edge that
                # the action names take from theirs -- one offset, so the panel's label and the
                # two below it read as one row of labels rather than three separate decisions.
                gk = " gold" if o_gold.get(cell) == label else ""
                parts.append('<text class="tsub%s" x="%.1f" y="%.1f" font-size="%.1f">%s</text>'
                             % (gk, px + pw / 2, panel_top + ACTION_DROP,
                                PANEL_LABEL_SIZE, label))

                parts.append('<rect class="panel" x="%.1f" y="%.1f" width="%.1f" height="%.1f"/>'
                             % (px, panel_top, pw, panel_h))

        # not a hard half, because the join is where the two scenes actually meet and the hover
        # halves are cut on the same number. Reading it means a tile that ever genuinely differs
        # moves its captions with its artwork instead of against it.

        if act_art and len(acts) == 2:
            mid = x0 + (x1 - x0) * joins.get(i, 0.5)
            a_top = panel_bot + ACT_TOP_GAP
            a_bot = y1 - PANEL_INSET
            boxes = (("left", x0 + PANEL_INSET, mid - ACT_SPLIT_GAP),
                     ("right", mid + ACT_SPLIT_GAP, x1 - PANEL_INSET))
            for half, bx0, bx1 in boxes:
                src = act_art.get(half)
                if src is None or not src.is_file():
                    raise SystemExit(
                        "action art for the %s half is missing (%s). The half would draw the "
                        "tile's plain fill and look deliberate." % (half, src))
                cid = "act%d%s" % (cell, half)
                bw, bh = bx1 - bx0, a_bot - a_top
                parts.append('<clipPath id="%s"><rect x="%.1f" y="%.1f" width="%.1f" '
                             'height="%.1f"/></clipPath>' % (cid, bx0, a_top, bw, bh))
                z = ACT_ZOOM.get(half, 1.0)
                dw, dh = bw * z, bh * z
                ox = bx0 - (dw - bw) / 2
                oy = a_top - (dh - bh) * ACT_PAN.get(half, 0.5)
                parts.append('<image href="%s" x="%.1f" y="%.1f" width="%.1f" height="%.1f" '
                             'preserveAspectRatio="xMidYMid slice" clip-path="url(#%s)"/>'
                             % (data_uri(src), ox, oy, dw, dh, cid))

        ink_l, ink_r = span_at(d, row_y)
        if len(acts) == 2:
            j = joins.get(i, 0.5)

                lo, hi = (ink_l, mid) if k == 0 else (mid, ink_r)
                size = min(22.0, (hi - lo - 18) / (len(act) * 0.46))
                gk = " gold" if o_gold.get(cell) == act else ""
                parts.append('<text class="tsub%s" x="%.1f" y="%.1f" font-size="%.1f">%s</text>'
                             % (gk, (lo + hi) / 2, row_y, size, act))
        elif acts:
            size = min(22.0, (ink_r - ink_l - 24) / (len(acts[0]) * 0.46))
            gk = " gold" if o_gold.get(cell) == acts[0] else ""

        line_y = panel_bot if in_panel else (base_y - LINE_ABOVE)

        # the acolyte line". On a tile whose acolytes have moved up into the panel the two ends
        # coincided and the line came out zero-length -- drawn, valid, and invisible.

        split_end = y1 if full else (base_y - LINE_ABOVE)

            parts.append('<line class="split" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>'

        bars = []
        for seat in (() if has_rects else BAR_SEATS.get(name, ())):
            j = dg.SEAT_ORDER.index(seat)
            if counts[j] > 0:
                bars.append((j, seat, counts[j]))
                counts[j] = 0                  # these seats are drawn below, not by the row

        # had been gated on "are the acolytes in the panel" alone, which is a different question:
        # the Ordination tile is not a panel tile and draws nothing, and got a rule across its
        # foot with nothing above it.

            parts.append('<line class="ground" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f"/>'

        for j, seat, n in bars:
            # ONE SEGMENT PER HIDDEN ACOLYTE, not one bar of their combined height. Each is
            # exactly the step a stacked figure would have shown -- fh * step -- so the column is
            # the same height either way and the white strokes between them are countable. That
            # is the whole gain: a solid bar says "more than one" and a segmented one says how
            # many, without going back to drawing a figure per acolyte.
            #
            # A single acolyte gets NO segments, which is right: a seat at 1 must look identical
            # whether it is drawn this way or stacked, or the two representations disagree at the
            # commonest count on the board.
            seg = fh * step
            bar_h = (n - 1) * seg
            bx = box_i["sx"] + j * (fw + gap)
            for k in range(n - 1):
                parts.append('<rect x="%.1f" y="%.1f" width="%.1f" height="%.1f" rx="%.1f" '
                             'fill="%s" stroke="%s" stroke-width="%.2f"/>'
                             % (bx, line_y - (k + 1) * seg, fw, seg, RECT_RADIUS,
                                pop.SEAT_SWATCH[seat], pop.HOOD_INK, fw * RECT_STROKE))
            parts.append('<use href="#%s-hood-%s" transform="translate(%.1f %.1f) scale(%.2f)"/>'
                         % (UID, seat, bx + fw / 2, line_y - bar_h - fh, fw))
        lit_seat, lit_at = o_litp.get(cell, (None, None))
        for seat, n in (squares or {}).items():
            j = dg.SEAT_ORDER.index(seat)
            sxp = box_i["sx"] + j * (fw + gap)
            for k in range(n):
                # The TOP one is the last drawn, k == n - 1: the stack grows upward from the line.
                base = pop.SEAT_SWATCH[seat]
                # k counts UP from the line, so k + 1 is the piece's position from the base --
                # the same way a player would count them.
                is_lit = (seat == lit_seat and (k + 1) == lit_at)
                fill = shade(base, LIT_V, LIT_S) if is_lit else shade(base, DIM_V, DIM_S)
                parts.append('<rect x="%.1f" y="%.1f" width="%.1f" height="%.1f" rx="%.1f" '
                             'fill="%s" stroke="%s" stroke-width="%.2f"/>'
                             % (sxp, line_y - (k + 1) * seg_h, fw, seg_h, RECT_RADIUS,
                                fill, pop.HOOD_INK, fw * RECT_STROKE))
        parts.append("</g>")
    parts.append("</svg>")

    css = (CSS.replace("__BG__", PAGE_BG).replace("__FILL__", FILL)
              .replace("__INK__", dg.INK).replace("__SW__", "%.2f" % (box * dg.EDGE_STROKE))

              .replace("__MARKO__", "%.2f" % MARK_OPACITY)
              .replace("__HALO__", "%.2f" % ACTION_HALO)
              .replace("__GOLD__", dg.EDGE_HOVER).replace("__MARKDUR__", MARK_DUR))
    cap = ("<b>The nine shapes, still</b>Outlines, fill, ink and stroke from gen_duty_grid; "
           "action names from gen_board.DUTY_TEXT, shortened. No animation, no art, nothing in "
           "the repository touched.")
    if plan is not PLAN_NINE and plan != PLAN_NINE:
        # DERIVED from the table, not written alongside it. Every rearrangement so far has left a
        # hand-written caption describing the arrangement before last, which is a small lie on a
        # page whose whole job is to be looked at.
        drawn = sorted({dg.DUTY_NAMES[x] for x in plan if x is not None})
        art = sum(1 for t in TILES if t["art"])
        gold = sum(1 for t in TILES if t["gold"])
        marked = sum(1 for t in TILES if t["mark"])
        cap = ("<b>%s &mdash; nine states</b>%d tile%s with pictures, %d of them with a label in "
               "the board's hover gold; %d marked. The rest carry pieces only, in the seat "
               "colours dimmed." % (", ".join(drawn), art, "" if art == 1 else "s", gold, marked))
    # The derived caption is "<b>title</b>rest"; split it so the title is a heading and the
    # rest is a line under it, rather than one bold run that reads as a sentence fused to a word.
    head, _, sub = cap.partition("</b>")
    notes = ["<h1>%s</h1>" % head.replace("<b>", ""),
             '<p style="color:#a99e86;font-size:12.3px;margin-top:-4px">%s</p>' % sub]
    notes += ["<p>%s</p>" % t for t in INTRO]
    for cells, head, body in NOTES:
        notes.append('<div class="n" data-cells="%s"><b>%s</b>%s</div>'
                     % (",".join(str(c) for c in cells), head, body))
    notes.append('<p class="foot"><b>Note:</b> %s</p>' % FOOTNOTE)

    return (PAGE.replace("__CSS__", css).replace("__NOTES__", "".join(notes)).replace("__SVG__", "".join(parts))
                .replace("__CAP__", cap))
